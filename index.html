viewof selected_flag = Inputs.select(flags, {
  label: "Select a country:",
  format: d => d.name,
  value: flags.find(d => d.code === "US")
})

// Euclidean distance between two flags' 27-bin color profiles
function colorDistance(a, b) {
  let sum = 0;
  for (let i = 0; i < a.colors.length; i++) {
    const diff = a.colors[i] - b.colors[i];
    sum += diff * diff;
  }
  return Math.sqrt(sum);
}

// Find the 6 countries with the most similar flag colors
similar_flags = flags
  .filter(d => d.file !== selected_flag.file)
  .map(d => ({ ...d, distance: colorDistance(selected_flag, d) }))
  .sort((a, b) => a.distance - b.distance)
  .slice(0, 6)

html`

       style="max-width: 480px; width: 100%; border: 1px solid #ccc; box-shadow: 0 2px 6px rgba(0,0,0,0.15);">
  <p style="margin-top: 0.6rem; font-size: 1.05rem;">
    <strong>National Anthem:</strong>
    <a href="https://www.youtube.com/results?search_query=${encodeURIComponent(selected_flag.name + ' national anthem ' + selected_flag.anthem)}"
       target="_blank" rel="noopener noreferrer"
       title="Click to listen to ${selected_flag.anthem}">
      <em>${selected_flag.anthem}</em> 🔊
    </a>
  </p>
</div>

<div style="margin-top: 2rem;">
